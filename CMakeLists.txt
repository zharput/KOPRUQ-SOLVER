cmake_minimum_required(VERSION 3.20)

project(KOPRUQ_SOLVER VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

find_package(Eigen3 CONFIG REQUIRED)

add_library(kopruq_solver_core INTERFACE)
target_include_directories(kopruq_solver_core INTERFACE ${PROJECT_SOURCE_DIR}/include)
target_link_libraries(kopruq_solver_core INTERFACE Eigen3::Eigen)
target_compile_features(kopruq_solver_core INTERFACE cxx_std_20)

if(MSVC)
  target_compile_options(kopruq_solver_core INTERFACE /W4 /permissive- /EHsc)
else()
  target_compile_options(kopruq_solver_core INTERFACE -Wall -Wextra -Wpedantic)
endif()

# JSON model/results I/O and the CLI front end; the solver core stays free of the JSON dependency.
find_package(nlohmann_json CONFIG REQUIRED)

target_link_libraries(kopruq_solver_io INTERFACE kopruq_solver_core nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(kopruq_solver_io INTERFACE /bigobj)
endif()

add_executable(kopruq_solver_cli apps/cli/main.cpp)
target_link_libraries(kopruq_solver_cli PRIVATE kopruq_solver_io)

target_link_libraries(kopruq_sparse_assembly_benchmark PRIVATE kopruq_solver_core)

target_link_libraries(kopruq_solver_benchmark PRIVATE kopruq_solver_core)

# Qt6 verification GUI (apps/verification-gui): off by default. The routine ci.yml matrix never
# sets KOPRUQ_BUILD_GUI, so qtbase is never requested from vcpkg and Ubuntu CI is unaffected;
# build with the "ninja-gui-release" preset (MSVC, VCPKG_MANIFEST_FEATURES=gui) to exercise it --
# see docs/SOLVER_ROADMAP.md's "QT6 VERIFICATION GUI" section for why this is a manual/dispatched
# build rather than part of every push.
option(KOPRUQ_BUILD_GUI "Build the Qt6 Widgets MIDAS verification GUI (apps/verification-gui)" OFF)
if(KOPRUQ_BUILD_GUI)
  find_package(Qt6 CONFIG REQUIRED COMPONENTS Widgets)

  # verification/: Qt-independent (GUI-4 Excel reading today; ResultMatcher/ComparisonEngine/
  # VerificationReportExporter land here in later milestones) -- gated the same way as the GUI
  # itself, since xlnt is only ever fetched via the "gui" vcpkg feature.
  find_package(xlnt CONFIG REQUIRED)
  add_library(kopruq_verification STATIC
    verification/ExcelReferenceReader.cpp
    verification/ComparisonEngine.cpp
    verification/ResultMatcher.cpp
    verification/VerificationReportExporter.cpp)
  target_include_directories(kopruq_verification PUBLIC ${PROJECT_SOURCE_DIR}/verification)
  target_link_libraries(kopruq_verification PUBLIC xlnt::xlnt nlohmann_json::nlohmann_json)
  target_compile_features(kopruq_verification PUBLIC cxx_std_20)
  if(MSVC)
    target_compile_options(kopruq_verification PRIVATE /bigobj)
  endif()

  set(CMAKE_AUTOMOC ON)
  add_executable(kopruq_verification_gui
    apps/verification-gui/main.cpp
    apps/verification-gui/MainWindow.cpp
    apps/verification-gui/FileInputPanel.cpp
    apps/verification-gui/AnalysisControlPanel.cpp
    apps/verification-gui/ExecutionLogPanel.cpp
    apps/verification-gui/ResultsComparisonPanel.cpp
    apps/verification-gui/SolverRunner.cpp)
  target_link_libraries(kopruq_verification_gui PRIVATE kopruq_solver_io Qt6::Widgets kopruq_verification)
  set(CMAKE_AUTOMOC OFF)
  add_custom_command(TARGET kopruq_verification_gui POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
            ${PROJECT_SOURCE_DIR}/apps/verification-gui/resources
            $<TARGET_FILE_DIR:kopruq_verification_gui>/resources)
  # A Qt Widgets app cannot even start without its platform plugin (qwindows.dll on Windows)
  # deployed next to the exe -- vcpkg's "applocal" step (used elsewhere in this project for DLL
  # dependencies) copies linked DLLs but not Qt plugins. Run windeployqt if vcpkg's Qt6 port
  # provides it; skip quietly if not found rather than failing the whole configure, since full
  # deployment tooling is GUI-7's scope -- GUI-1 only needs this for the CI launch smoke test.
  find_program(KOPRUQ_WINDEPLOYQT windeployqt HINTS "${CMAKE_PREFIX_PATH}" "${Qt6_DIR}/../../../tools/Qt6/bin")
  if(WIN32 AND KOPRUQ_WINDEPLOYQT)
    add_custom_command(TARGET kopruq_verification_gui POST_BUILD
      COMMAND ${KOPRUQ_WINDEPLOYQT} --no-translations --no-compiler-runtime $<TARGET_FILE:kopruq_verification_gui>)
  endif()
endif()
